% Sprint 1 Use Case Diagram - Authentication & RBAC
% UCs: Authenticate, Reset Password, Assign Role
% ============================================================
\begin{tikzpicture}[
  syst/.style={rectangle, draw=darkGray, fill=gray!10, sharp corners,
               minimum width=2.4cm, minimum height=0.6cm, align=center,
               font=\scriptsize\bfseries},
  uc/.style={ellipse, draw=isiBlue!70, line width=0.7pt, fill=blue!4,
             minimum width=3.4cm, minimum height=0.6cm,
             align=center, font=\footnotesize, inner sep=1pt},
  gen/.style={-{Triangle[open, length=4mm, width=4mm]}, line width=0.7pt, draw=darkGray},
  inc/.style={-Latex, dashed, draw=isiBlue!70, font=\scriptsize\itshape},
  ext/.style={-Latex, dashed, draw=isiBlue!70, font=\scriptsize\itshape},
  assoc/.style={draw=darkGray!70, line width=0.5pt},
  boundary/.style={draw=isiBlue, very thick, fill=gray!2, rounded corners=8pt}
]

  \draw[boundary] (-4.2,-4.0) rectangle (4.2,3.6);
  \node[font=\small\bfseries\color{isiBlue}, anchor=north]
        at (0,3.5) {«system» \ Sprint 1};

  % Use cases (centred)
  \node[uc] (register)     at (0, 2.6)  {Register};
  \node[uc] (authenticate) at (0, 1.5)  {Authenticate};
  \node[uc] (oauth)        at (0, 0.4)  {OAuth Login};
  \node[uc] (reset)        at (0,-0.7)  {Reset Password};
  \node[uc] (profile)      at (0,-1.8)  {Manage Profile};
  \node[uc] (assign)       at (0,-2.9)  {Assign Role};

  % Primary actors (LEFT, stick figures)
  \umlactor{visitor}{-6.5}{ 2.0}{Visitor}
  \umlactor{member} {-6.5}{-0.7}{Member}
  \umlactor{admin}  {-6.5}{-2.9}{Administrator}

  % Actor generalization (child --|> parent)
  \draw[gen] (member.north) -- (visitor.south);
  \draw[gen] (admin.north)  -- (member.south);

  % Secondary actor (RIGHT)
  \node[syst] (oauthSys) at (6.5, 0.4) {«system»\\OAuth Provider};

  % Associations
  \draw[assoc] (visitor) -- (register);
  \draw[assoc] (visitor) -- (authenticate);
  \draw[assoc] (visitor) -- (reset);
  \draw[assoc] (member)  -- (profile);
  \draw[assoc] (admin)   -- (assign);

  % External system link
  \draw[assoc] (oauth.east) -- (oauthSys.west);

  % «extend» — OAuth extends Authenticate
  \draw[ext] (oauth.north) to[bend left=10] node[right, font=\tiny] {«extend»} (authenticate.south);

  % «include» — Profile and Assign Role require Authenticate
  \draw[inc] (profile.east) to[bend left=18] node[right, font=\tiny] {«include»} (authenticate.east);
  \draw[inc] (assign.east)  to[bend left=28] (authenticate.east);

\end{tikzpicture}
