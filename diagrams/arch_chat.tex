\begin{tikzpicture}[
  client/.style={rectangle, draw=isiBlue, fill=blue!8, rounded corners=4pt,
                 minimum width=3cm, minimum height=1cm, align=center,
                 font=\footnotesize\bfseries},
  server/.style={rectangle, draw=orange, fill=orange!10, rounded corners=4pt,
                 minimum width=4.5cm, minimum height=1.2cm, align=center,
                 font=\small\bfseries, text=orange!80!black},
  db/.style={rectangle, draw=darkGray, fill=gray!8, rounded corners=3pt,
             minimum width=3cm, minimum height=0.9cm, align=center,
             font=\footnotesize\bfseries},
  room/.style={rectangle, draw=igsGreen, fill=green!6, rounded corners=3pt,
               minimum width=3cm, minimum height=0.8cm, align=center,
               font=\tiny},
  arr/.style={->, thick, draw=isiBlue},
  arrsio/.style={<->, thick, draw=orange},
  lbl/.style={font=\tiny, midway},
  note/.style={draw=gray!40, fill=gray!5, rounded corners=2pt, font=\tiny, inner sep=3pt}]

  % ---- Clients ----
  \node[client] (c1) at (-5.5, 3) {Client A\\(Workspace App)};
  \node[client] (c2) at (-5.5, 1) {Client B\\(Workspace App)};
  \node[client] (c3) at (-5.5,-1) {Client C\\(DM only)};

  % ---- Socket.io Server ----
  \node[server] (sio) at (0, 1) {Socket.io Server\\(port 3001)};

  % ---- Rooms ----
  \node[room] (r1) at (0,  3.5) {Room: channel:\{id\}};
  \node[room] (r2) at (0, -0.5) {Room: dm:\{uid1\}:\{uid2\}};

  % ---- DB ----
  \node[db] (db)  at (5, 2) {PostgreSQL\\chat\_messages};
  \node[db] (db2) at (5, 0) {PostgreSQL\\direct\_messages};

  % ---- Arrows: Clients $\leftrightarrow$ Socket.io ----
  \draw[arrsio] (c1.east) -- node[lbl, above] {join\_room / send\_message} (sio.west |- c1);
  \draw[arrsio] (c2.east) -- node[lbl, above] {join\_room / send\_message} (sio.west |- c2);
  \draw[arrsio] (c3.east) -- node[lbl, above] {join\_room / send\_dm} (sio.west |- c3);

  % ---- Socket.io → Rooms ----
  \draw[arr, orange] (sio.north) -- node[lbl, right] {broadcast} (r1.south);
  \draw[arr, orange] (sio.south) -- node[lbl, right] {broadcast} (r2.north);

  % ---- Server → DB ----
  \draw[arr] (sio.east) -- node[lbl, above] {persist} (db.west);
  \draw[arr] (sio.east) -- node[lbl, below] {persist} (db2.west);

  % ---- Events legend ----
  \node[note, align=left] at (0,-3.0) {
    \textbf{Socket Events (client $\rightarrow$ server):}\\
    \texttt{join\_room} -- subscribe to a channel room\\
    \texttt{send\_message} -- emit message to channel\\
    \texttt{send\_dm} -- emit private message\\
    \texttt{add\_reaction} -- add emoji reaction to a message\\
    \texttt{mark\_read} -- mark messages as read\\[4pt]
    \textbf{Socket Events (server $\rightarrow$ clients):}\\
    \texttt{new\_message} -- broadcast to channel room members\\
    \texttt{new\_dm} -- emit to DM room\\
    \texttt{reaction\_update} -- updated reactions for a message\\
    \texttt{user\_online} -- presence status update
  };

  % ---- Message persistence note ----
  \node[note, right, align=left] at (5.2, 3.5) {
    Message history\\loaded from DB\\on channel join
  };
  \draw[->, gray, dashed] (db.north) -- (5, 3.3);

\end{tikzpicture}
