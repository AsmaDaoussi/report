% ============================================================
% Deployment Diagram - Proposed Production Stack
%
% draw.io equivalent (UML deployment notation):
%   Dev workstation -- push --> GitHub repo
%   GitHub repo -- triggers --> GitHub Actions
%   GitHub Actions -- builds --> Docker images (3 apps)
%   Docker images -- pushed to --> Docker Hub
%   Docker Hub <-- pulled by --> Vercel (marketplace, workspace), Render (backend)
%   Vercel (marketplace) <--HTTPS--> External user browser
%   Vercel (workspace)   <--HTTPS--> Internal user browser
%   Render (backend)     <--SQL-->  Supabase Postgres
%   Render (backend)     <--SDK-->  Supabase Storage
%   Render (backend)     <--HTTPS--> Stripe API
%   Render (backend)     <--HTTPS--> HuggingFace Inference
%   Sentry collects errors from all three apps
% ============================================================
\begin{tikzpicture}[
  node distance=1cm,
  box/.style={rectangle, draw=isiBlue, fill=blue!4, rounded corners=4pt,
              minimum width=2.5cm, minimum height=0.9cm, align=center,
              font=\scriptsize\bfseries},
  cloud/.style={rectangle, draw=darkGray, fill=gray!8, rounded corners=6pt,
                minimum width=2.8cm, minimum height=0.9cm, align=center,
                font=\scriptsize\bfseries},
  user/.style={rectangle, draw=isiBlue, fill=blue!12, rounded corners=3pt,
               minimum width=2.2cm, minimum height=0.7cm, align=center,
               font=\scriptsize\bfseries},
  arr/.style={-Latex, thick, draw=isiBlue},
  dashed_arr/.style={-Latex, thick, draw=darkGray, dashed},
  lbl/.style={font=\tiny, midway, fill=white, inner sep=1pt}]

  % ============ DEVELOPMENT LAYER (top) ============
  \node[user]  (dev)    at (-5, 5.5) {Developer\\workstation};
  \node[cloud] (github) at ( 0, 5.5) {GitHub\\Repository};
  \node[cloud] (gha)    at ( 5, 5.5) {GitHub\\Actions};

  \draw[arr] (dev)    -- node[lbl] {git push} (github);
  \draw[arr] (github) -- node[lbl] {trigger}  (gha);

  % ============ BUILD LAYER ============
  \node[cloud] (hub) at ( 5, 3.5) {Docker Hub};
  \draw[arr] (gha) -- node[lbl] {push images} (hub);

  % ============ HOSTING LAYER ============
  \node[box] (vercelM) at (-4, 1.5) {Vercel\\Marketplace};
  \node[box] (vercelW) at (-1, 1.5) {Vercel\\Workspace};
  \node[box] (render)  at ( 3.5, 1.5) {Render\\Express API};

  \draw[arr] (hub) -- node[lbl, sloped] {pull image} (vercelM.north);
  \draw[arr] (hub) -- node[lbl]         {pull image} (vercelW.north);
  \draw[arr] (hub) -- node[lbl]         {pull image} (render.north);

  % ============ DATA LAYER ============
  \node[cloud] (pg)      at (-2,-1.2) {Supabase\\PostgreSQL};
  \node[cloud] (storage) at ( 2,-1.2) {Supabase\\Storage};
  \node[cloud] (stripe)  at ( 5.8,-0.5) {Stripe API};
  \node[cloud] (hf)      at ( 5.8,-1.9) {HuggingFace\\Inference};

  \draw[arr] (render) -- node[lbl] {SQL} (pg);
  \draw[arr] (render) -- node[lbl] {SDK} (storage);
  \draw[arr] (render.east) to[bend left=10]  node[lbl] {HTTPS} (stripe);
  \draw[arr] (render.east) to[bend right=10] node[lbl] {HTTPS} (hf);

  % ============ USER LAYER (bottom) ============
  \node[user] (publicUser)   at (-4,-3.5) {External User};
  \node[user] (internalUser) at (-1,-3.5) {Internal User};

  \draw[arr] (publicUser)   -- node[lbl] {HTTPS} (vercelM);
  \draw[arr] (internalUser) -- node[lbl] {HTTPS} (vercelW);

  % ============ MONITORING ============
  \node[cloud] (sentry) at ( 6.5, 3.5) {Sentry\\(monitoring)};
  \draw[dashed_arr] (vercelM.north east) to[bend right=20] (sentry.west);
  \draw[dashed_arr] (vercelW.north east) to[bend right=10] (sentry.west);
  \draw[dashed_arr] (render.north) -- (sentry.south);

\end{tikzpicture}
