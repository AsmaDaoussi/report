% Stripe Payment Sequence Diagram
\begin{tikzpicture}[
  part/.style={rectangle, draw=isiBlue, fill=blue!8, minimum width=2.6cm,
               minimum height=0.65cm, align=center, font=\footnotesize\bfseries},
  msg/.style={->, draw=isiBlue, thick},
  ret/.style={->, draw=gray, dashed},
  lbl/.style={font=\tiny, midway, above=1pt},
  note/.style={draw=orange!70, fill=orange!10, rounded corners=2pt,
               font=\tiny, inner sep=3pt, align=left}]

  % ---- Participants ----
  \node[part] (buyer)   at (0,    0) {Buyer (Browser)};
  \node[part] (backend) at (4,    0) {Express Backend};
  \node[part] (stripe)  at (8,    0) {Stripe API};
  \node[part] (db)      at (12,   0) {PostgreSQL};

  % ---- Lifelines ----
  \foreach \x in {0,4,8,12}
    \draw[dashed, gray!60] (\x,-0.32) -- (\x,-13.5);

  % ---- Activation Bars ----
  \filldraw[fill=blue!15, draw=isiBlue!50] (-0.15,-0.32) rectangle (0.15,-13.0);
  \filldraw[fill=blue!15, draw=isiBlue!50] (3.85, -1.0)  rectangle (4.15,  -4.5);
  \filldraw[fill=blue!15, draw=isiBlue!50] (7.85, -2.0)  rectangle (8.15,  -3.5);
  \filldraw[fill=blue!15, draw=isiBlue!50] (3.85, -5.5)  rectangle (4.15,  -11.5);
  \filldraw[fill=blue!15, draw=isiBlue!50] (11.85,-7.5)  rectangle (12.15, -11.0);

  % ---- Phase 1: Create PaymentIntent ----
  \node[note, align=center] at (-2.5,-1.0) {\textbf{Phase 1:}\\Create Intent};
  \draw[msg] (0.15,-1.0)  -- node[lbl] {POST /api/create-payment-intent \{cartItems\}} (3.85,-1.0);
  \draw[msg] (4.15,-2.0)  -- node[lbl] {paymentIntents.create(\{amount, currency\})} (7.85,-2.0);
  \draw[ret] (7.85,-3.5)  -- node[lbl] {PaymentIntent \{clientSecret\}} (4.15,-3.5);
  \draw[ret] (3.85,-4.5)  -- node[lbl] {200 \{clientSecret\}} (0.15,-4.5);

  % ---- Phase 2: Client confirms payment ----
  \node[note, align=center] at (-2.5,-5.5) {\textbf{Phase 2:}\\Confirm};
  \node[note, right] at (0.2,-5.3) {Stripe.js: confirmCardPayment(clientSecret, \{card\})};
  \draw[msg] (0.15,-5.5)  -- node[lbl] {confirmCardPayment(clientSecret, cardElement)} (7.85,-5.5);
  \draw[ret] (7.85,-6.5)  -- node[lbl] {\{paymentIntent: succeeded\}} (0.15,-6.5);

  % ---- Phase 3: Record order ----
  \node[note, align=center] at (-2.5,-7.5) {\textbf{Phase 3:}\\Record Order};
  \draw[msg] (0.15,-7.5)  -- node[lbl] {POST /api/orders \{paymentIntentId, cartItems\}} (3.85,-7.5);
  \draw[msg] (4.15,-8.5)  -- node[lbl] {BEGIN TRANSACTION} (11.85,-8.5);
  \draw[msg] (4.15,-9.2)  -- node[lbl] {INSERT INTO orders (status='paid')} (11.85,-9.2);
  \draw[msg] (4.15,-9.9)  -- node[lbl] {INSERT INTO order\_items} (11.85,-9.9);
  \draw[msg] (4.15,-10.6) -- node[lbl] {INSERT INTO user\_downloads (unlock assets)} (11.85,-10.6);
  \draw[ret] (11.85,-11.0) -- node[lbl] {COMMIT} (4.15,-11.0);

  % Notify
  \node[note, right] at (4.2,-11.5) {createNotification(buyer, `order\_confirmed')};

  \draw[ret] (3.85,-12.0) -- node[lbl] {201 \{order, downloads unlocked\}} (0.15,-12.0);

  % ---- Termination ----
  \foreach \x in {0,4,8,12}
    \draw[thick] (\x-0.2,-13.2) -- (\x+0.2,-13.2);
