% ER Diagram -- Sprint 1: Users, Roles, Permissions, Audit
\begin{tikzpicture}[
  hdr/.style={rectangle, draw=isiBlue, fill=isiBlue, text=white,
              minimum width=3.8cm, minimum height=0.55cm,
              align=center, font=\footnotesize\bfseries},
  attr/.style={rectangle, draw=isiBlue!40, fill=white,
               minimum width=3.8cm, minimum height=0.45cm,
               align=left, font=\ttfamily\tiny, inner sep=3pt},
  pk/.style={attr, font=\ttfamily\tiny\bfseries},
  fk/.style={attr, font=\ttfamily\tiny\itshape, text=igsGreen},
  rel/.style={draw=isiBlue!70, thick},
  card/.style={font=\tiny\bfseries\color{isiBlue}}]

  % =================== users ===================
  \node[hdr] (u_hdr) at (0,0) {users};
  \node[pk,  below=0pt of u_hdr] (u1) {id : UUID (PK)};
  \node[attr, below=0pt of u1]   (u2) {firebase\_uid : VARCHAR};
  \node[attr, below=0pt of u2]   (u3) {email : VARCHAR};
  \node[attr, below=0pt of u3]   (u4) {username : VARCHAR};
  \node[attr, below=0pt of u4]   (u5) {user\_type : ENUM};
  \node[attr, below=0pt of u5]   (u6) {status : ENUM};
  \node[attr, below=0pt of u6]   (u7) {is\_seller : BOOLEAN};
  \node[attr, below=0pt of u7]   (u8) {created\_at : TIMESTAMPTZ};

  % =================== user_roles ===================
  \node[hdr] (ur_hdr) at (5.5, 0) {user\_roles};
  \node[pk,  below=0pt of ur_hdr] (ur1) {id : SERIAL (PK)};
  \node[fk,  below=0pt of ur1]   (ur2) {user\_id : UUID (FK->users)};
  \node[fk,  below=0pt of ur2]   (ur3) {role\_id : INT (FK->roles)};
  \node[attr, below=0pt of ur3]  (ur4) {assigned\_at : TIMESTAMPTZ};
  \node[attr, below=0pt of ur4]  (ur5) {assigned\_by : UUID};

  % =================== roles ===================
  \node[hdr] (r_hdr) at (10.5, 0) {roles};
  \node[pk,  below=0pt of r_hdr] (r1) {id : SERIAL (PK)};
  \node[attr, below=0pt of r1]   (r2) {name : VARCHAR};
  \node[attr, below=0pt of r2]   (r3) {display\_name : VARCHAR};
  \node[attr, below=0pt of r3]   (r4) {level : INTEGER};
  \node[attr, below=0pt of r4]   (r5) {user\_type : ENUM};

  % =================== role_permissions ===================
  \node[hdr] (rp_hdr) at (10.5, -4.5) {role\_permissions};
  \node[pk,  below=0pt of rp_hdr] (rp1) {id : SERIAL (PK)};
  \node[fk,  below=0pt of rp1]   (rp2) {role\_id : INT (FK->roles)};
  \node[attr, below=0pt of rp2]  (rp3) {permission : VARCHAR};

  % =================== audit_logs ===================
  \node[hdr] (al_hdr) at (0, -4.5) {audit\_logs};
  \node[pk,  below=0pt of al_hdr] (al1) {id : UUID (PK)};
  \node[fk,  below=0pt of al1]   (al2) {user\_id : UUID (FK->users)};
  \node[attr, below=0pt of al2]  (al3) {action : VARCHAR};
  \node[attr, below=0pt of al3]  (al4) {resource\_type : VARCHAR};
  \node[attr, below=0pt of al4]  (al5) {resource\_id : VARCHAR};
  \node[attr, below=0pt of al5]  (al6) {details : JSONB};
  \node[attr, below=0pt of al6]  (al7) {created\_at : TIMESTAMPTZ};

  % =================== Relationships ===================
  % users 1--* user_roles
  \draw[rel] (u_hdr.east)  -- node[card, above] {1} ++(0.3,0)
                             -- node[card, above] {*} (ur_hdr.west);
  % user_roles *--1 roles
  \draw[rel] (ur_hdr.east) -- node[card, above] {*} ++(0.3,0)
                             -- node[card, above] {1} (r_hdr.west);
  % roles 1--* role_permissions
  \draw[rel] (r_hdr.south) -- node[card, right] {1} ++(0,-0.3)
                             |- node[card, above, near end] {*} (rp_hdr.north);
  % users 1--* audit_logs
  \draw[rel] (u8.south) -- node[card, right] {1} ++(0,-0.2)
                          |- node[card, above, near end] {*} (al_hdr.north);

\end{tikzpicture}
