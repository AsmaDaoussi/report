% Admin Moderation Use Case Diagram
\begin{tikzpicture}[
  actor/.style={rectangle, draw=isiBlue, fill=blue!10, rounded corners=3pt,
                minimum width=1.8cm, minimum height=0.65cm, align=center,
                font=\footnotesize\bfseries},
  uc/.style={ellipse, draw=isiBlue!60, fill=blue!4, minimum width=4cm,
             minimum height=0.65cm, align=center, font=\footnotesize, inner sep=1pt},
  inc/.style={draw=gray, dashed, ->, font=\tiny\itshape},
  assoc/.style={draw=gray!70, thin},
  boundary/.style={draw=isiBlue, very thick, fill=gray!3, rounded corners=6pt}]

  % ---- System Boundary ----
  \draw[boundary] (-3.8,-7.2) rectangle (4.2,2.0);
  \node[font=\footnotesize\bfseries\color{isiBlue}] at (0.2,1.8)
    {$\ll$subsystem$\gg$ Admin Panel};

  % ---- Use Cases ----
  \node[uc] (viewpending) at (0, 1.2)  {View Pending Assets};
  \node[uc] (approve)     at (0, 0.2)  {Approve Asset};
  \node[uc] (reject)      at (0,-0.8)  {Reject Asset (with reason)};
  \node[uc] (viewusers)   at (0,-1.8)  {View All Users};
  \node[uc] (banuser)     at (0,-2.8)  {Ban / Activate User};
  \node[uc] (changerole)  at (0,-3.8)  {Change User Role};
  \node[uc] (viewanalytics) at (0,-4.8) {View Platform Analytics};
  \node[uc] (sellerreq)  at (0,-5.8)  {Manage Seller Requests};
  \node[uc] (audit)       at (0,-6.8)  {View Audit Logs};

  % ---- Actors ----
  \node[actor] (admin)    at (-6.5,-2.8) {Admin};
  \node[actor] (superadm) at (-6.5,-5.8) {Super Admin};
  \node[actor] (seller)   at (7.0,  0.2) {Seller};
  \node[actor] (system)   at (7.0, -3.8) {System\\(Automated)};

  % ---- Associations: Admin ----
  \draw[assoc] (admin) -- (viewpending);
  \draw[assoc] (admin) -- (approve);
  \draw[assoc] (admin) -- (reject);
  \draw[assoc] (admin) -- (viewusers);
  \draw[assoc] (admin) -- (banuser);
  \draw[assoc] (admin) -- (viewanalytics);
  \draw[assoc] (admin) -- (sellerreq);
  \draw[assoc] (admin) -- (audit);

  % ---- Associations: Super Admin ----
  \draw[assoc] (superadm) -- (changerole);
  \draw[assoc] (superadm) -- (banuser);
  \draw[inc] (superadm) -- node[below,font=\tiny] {$\ll$extends$\gg$} (admin);

  % ---- Associations: Seller (notified) ----
  \draw[assoc, dashed] (seller) -- node[above,font=\tiny] {receives notification} (approve);
  \draw[assoc, dashed] (seller) -- node[below,font=\tiny] {receives notification} (reject);

  % ---- Include relationships ----
  \draw[inc] (approve)   -- node[right, font=\tiny] {$\ll$include$\gg$ Audit Log} (audit);
  \draw[inc] (reject)    -- node[right, font=\tiny] {$\ll$include$\gg$ Audit Log} (audit);
  \draw[inc] (banuser)   -- node[right, font=\tiny] {$\ll$include$\gg$ Audit Log} (audit);
  \draw[inc] (changerole) -- node[right, font=\tiny] {$\ll$include$\gg$ Audit Log} (audit);

  % ---- System (automated) ----
  \draw[assoc] (system) -- (audit);

\end{tikzpicture}
